{%- comment -%} A list of post cards. Every card comes from post-card.html, so the only differences between lists are card size and the number of columns. posts: required. limit: optional max number of cards. lead: when true, the first card is wide and loads first (page 1). compact: small cards in three columns (Keep reading, 404). feature: one wide card per row (Read next). priority: when true, the first card's photo loads first. {%- endcomment -%} {%- assign card_posts = include.posts -%} {%- if include.limit -%} {%- assign card_limit = include.limit | plus: 0 -%} {%- assign card_posts = include.posts | slice: 0, card_limit -%} {%- endif -%}
{%- for post in card_posts %} {%- assign card_priority = false -%} {%- if forloop.first and include.lead or forloop.first and include.priority -%} {%- assign card_priority = true -%} {%- endif -%} {%- if include.compact -%} {%- include post-card.html post=post style="small" priority=card_priority -%} {%- elsif include.feature -%} {%- include post-card.html post=post style="wide" priority=card_priority -%} {%- elsif include.lead and forloop.first -%} {%- include post-card.html post=post style="wide" lead=true priority=card_priority -%} {%- else -%} {%- include post-card.html post=post priority=card_priority -%} {%- endif -%} {%- endfor %}
